-- Q1. Write a PL/SQL block to print first n natural numbers and their sum.
DO $$
DECLARE
    n INT := 10;
    i INT := 1;
    total INT := 0;
BEGIN
    WHILE i <= n LOOP
        RAISE NOTICE '%', i;
        total := total + i;
        i := i + 1;
    END LOOP;
    RAISE NOTICE 'Sum of first % natural numbers = %', n, total;
END $$;

-- Q2. Write a PL/SQL block to find out sum of the digits of a given number.
DO $$
DECLARE
    num INT := 12345;
    temp INT;
    digit_sum INT := 0;
BEGIN
    temp := num;
    WHILE temp > 0 LOOP
        digit_sum := digit_sum + (temp % 10);
        temp := temp / 10;
    END LOOP;
    RAISE NOTICE 'Sum of digits of % = %', num, digit_sum;
END $$;

-- Q3. Write a PL/SQL block to generate a Fibonacci series.
DO $$
DECLARE
    n INT := 10;
    a INT := 0;
    b INT := 1;
    nxt INT;
    i INT;
BEGIN
    RAISE NOTICE '%', a;
    RAISE NOTICE '%', b;
    FOR i IN 3..n LOOP
        nxt := a + b;
        RAISE NOTICE '%', nxt;
        a := b;
        b := nxt;
    END LOOP;
END $$;

-- Q4. Write a PL/SQL script that traps ZERO_DIVIDE exceptions.
DO $$
DECLARE
    a INT := 10;
    b INT := 0;
    result INT;
BEGIN
    result := a / b;
    RAISE NOTICE 'Result = %', result;
EXCEPTION
    WHEN division_by_zero THEN
        RAISE NOTICE 'Error: division by zero is not allowed';
END $$;

-- Q5. Write a PL/SQL block to illustrate functioning of IF_THEN_ELSIF by creating the equivalent of a four-function calculator.
DO $$
DECLARE
    num1 NUMERIC := 20;
    num2 NUMERIC := 4;
    op CHAR(1) := '*';
    result NUMERIC;
BEGIN
    IF op = '+' THEN
        result := num1 + num2;
    ELSIF op = '-' THEN
        result := num1 - num2;
    ELSIF op = '*' THEN
        result := num1 * num2;
    ELSIF op = '/' THEN
        result := num1 / num2;
    ELSE
        RAISE NOTICE 'Invalid operator';
    END IF;
    RAISE NOTICE '% % % = %', num1, op, num2, result;
END $$;

-- Q6. Write a PL/SQL Procedure to find the area of the circle.
CREATE OR REPLACE PROCEDURE URK24CS1021_circle_area(radius NUMERIC)
LANGUAGE plpgsql AS $$
DECLARE
    pi_val CONSTANT NUMERIC := 3.14159265358979;
    area NUMERIC;
BEGIN
    area := pi_val * radius * radius;
    RAISE NOTICE 'Area of circle with radius % = %', radius, round(area, 2);
END;
$$;
CALL URK24CS1021_circle_area(7);

-- Q7. Write a PL/SQL Function to find the area of the rectangle.
CREATE OR REPLACE FUNCTION URK24CS1021_rectangle_area(length NUMERIC, width NUMERIC)
RETURNS NUMERIC LANGUAGE plpgsql AS $$
DECLARE
    area NUMERIC;
BEGIN
    area := length * width;
    RETURN area;
END;
$$;
SELECT URK24CS1021_rectangle_area(5, 10) AS rectangle_area;
