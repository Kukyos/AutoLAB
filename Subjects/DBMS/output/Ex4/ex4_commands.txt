-- Q1. Display the cities of the companies Acc and TATA.
SELECT city FROM URK24CS1021_company WHERE cname = 'Acc'
UNION
SELECT city FROM URK24CS1021_company WHERE cname = 'TATA';

-- Q2. Display the names of the city that is common to companies Acc and TATA.
SELECT city FROM URK24CS1021_company WHERE cname = 'Acc'
INTERSECT
SELECT city FROM URK24CS1021_company WHERE cname = 'TATA';

-- Q3. Find the cities where Acc is located, but not TATA.
SELECT city FROM URK24CS1021_company WHERE cname = 'Acc'
EXCEPT
SELECT city FROM URK24CS1021_company WHERE cname = 'TATA';

-- Q4. Display the names of employees living in Nagpur, and working with company Acc.

-- Q6. List the names of the employees living in city Bombay and having salary >1500
SELECT e.ename FROM URK24CS1021_employee e, URK24CS1021_emp_company c
WHERE e.ename = c.ename AND e.city = 'Bombay' AND c.salary > 1500;

-- Q7. Display the names of the employees living in the city Chennai and working with the company CMC.

-- Q12. Display the employee's name whose joining date is 18-DEC-98 and whose working shift is 'E'.

WHERE ec.ename = s.ename AND ec.jdate = '1998-12-18' AND s.shift = 'E';

-- Q13. Use in operator to display the names of the employees living in the same city where Sunil is residing.
SELECT ename FROM URK24CS1021_employee
WHERE city IN (SELECT city FROM URK24CS1021_employee WHERE ename = 'Sunil');

-- Q14. Use in operator to display the employees living in the city Bombay and having their company located in the city Delhi.
SELECT ename FROM URK24CS1021_employee
WHERE city = 'Bombay' AND ename IN
    (SELECT ename FROM URK24CS1021_emp_company WHERE cname IN
        (SELECT cname FROM URK24CS1021_company WHERE city = 'Delhi'));
